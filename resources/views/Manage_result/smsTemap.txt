<script>
    document.addEventListener('DOMContentLoaded', function() {
        const studentSearch = document.getElementById('studentQuickSearch');
        const rows = document.querySelectorAll('.student-row-item');
        const checkboxes = document.querySelectorAll('.row-cb');
        const masterSelect = document.getElementById('masterSelect');
        const timeInput = document.getElementById('time_input');

        // 1. Live Filter logic
        studentSearch?.addEventListener('keyup', function() {
            let val = this.value.toLowerCase();
            rows.forEach(row => {
                row.style.display = row.getAttribute('data-search').includes(val) ? "block" : "none";
            });
        });

        // 2. Dynamic Preview Modal
        document.querySelectorAll('.view-msg-btn').forEach(btn => {
            btn.addEventListener('click', function() {
                let parent = this.closest('.student-row-item');
                document.getElementById('modalTitle').innerText = "Preview: " + parent.getAttribute('data-name');
                document.getElementById('modalBody').innerText = parent.getAttribute('data-message');
                new bootstrap.Modal(document.getElementById('previewModal')).show();
            });
        });

        // 3. Selection and Validation
        masterSelect?.addEventListener('change', function() {
            checkboxes.forEach(cb => {
                if(cb.closest('.student-row-item').style.display !== 'none') cb.checked = this.checked;
            });
            updateInterface();
        });

        checkboxes.forEach(cb => cb.addEventListener('change', updateInterface));
        timeInput.addEventListener('change', updateInterface);

        function updateInterface() {
            let count = 0, totalCost = 0;
            const balance = parseInt(document.getElementById('db-balance-val').innerText);
            
            checkboxes.forEach(cb => {
                if(cb.checked) {
                    count++;
                    totalCost += parseInt(cb.closest('.student-row-item').getAttribute('data-units'));
                }
            });

            const remaining = balance - totalCost;
            document.getElementById('ui-selected-count').innerText = count;
            document.getElementById('ui-units-cost').innerText = totalCost;
            
            const remLabel = document.getElementById('ui-remaining-bal');
            const alertBox = document.getElementById('balance-alert');
            const submitBtn = document.getElementById('mainSubmitBtn');

            const hasTime = timeInput.value !== "";

            if (count > 0) {
                remLabel.innerText = remaining;
                if (remaining < 0) {
                    remLabel.className = "fw-bold text-danger";
                    alertBox.style.display = 'block';
                    submitBtn.disabled = true;
                    submitBtn.classList.replace('btn-primary', 'btn-danger');
                } else {
                    remLabel.className = "fw-bold text-success";
                    alertBox.style.display = 'none';
                    // Button only activates if students are selected AND time is set
                    submitBtn.disabled = !hasTime;
                    submitBtn.classList.replace('btn-danger', 'btn-primary');
                }
            } else {
                remLabel.innerText = "---";
                submitBtn.disabled = true;
                alertBox.style.display = 'none';
            }
        }
    });

    function handleFinalDispatch() {
        const date = document.getElementById('date_input').value;
        const time = document.getElementById('time_input').value;

        if (!date || !time) {
            alert('Selection incomplete. Please pick both date and time.');
            return;
        }

        document.getElementById('hidden_date').value = date;
        document.getElementById('hidden_time').value = time;

        if(confirm(`Schedule result broadcast?`)) {
            const btn = document.getElementById('mainSubmitBtn');
            btn.disabled = true;
            btn.innerHTML = `<span class="spinner-border spinner-border-sm me-2"></span> Dispatching...`;
            document.getElementById('smsMasterForm').submit();
        }
    }
</script>
